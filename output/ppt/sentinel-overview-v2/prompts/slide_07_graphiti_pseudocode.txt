def retrieve_context(event):
    # 以当前事件作为查询输入
    graph_hits = graphiti.hybrid_search(
        query=event,
        methods=["vector_similarity", "BM25", "graph_walk"],
    )

    # 从图谱检索结果中取出关系边和历史事件片段
    edge_contexts = graph_hits.edges
    episode_contexts = graph_hits.episodes

    # 合并为候选上下文
    candidate_contexts = edge_contexts + episode_contexts

    # 根据当前事件相关性进行重排序和筛选
    relevant_contexts = reranker.rank_and_filter(
        query=event,
        candidates=candidate_contexts,
    )

    # 返回确实相关的图谱上下文
    return relevant_contexts
